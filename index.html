⑮ 链式 Builder API Service

HE.build().init().text().style().print() 链式调用，更优雅
const result = await HE.build('我的订单')
  .page(3, 80, 0)               // 80mm 卷筒
  .printer('XP-80C')            // 选打印机
  .text(20, 10, 280, 30, '订单 #001')
    .style('FontSize', 14)
    .style('Bold', 1)
    .style('Alignment', 2)
  .barcode(60, 50, 200, 200, 'QRCode', 'https://...')
  .line(150, 0, 150, 800, 'dashed', 1)
  .print(true);                  // true=静默

console.log(result);
